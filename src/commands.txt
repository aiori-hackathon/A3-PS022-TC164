TO HAVE A RESOLVER ON THE SERVER
sudo ss -tulpn | grep :53
sudo systemctl disable --now systemd-resolved
sudo rm /etc/resolv.conf
echo "nameserver 8.8.8.8" | sudo tee /etc/resolv.conf
sudo apt update && sudo apt install -y dnsmasq
sudo systemctl enable --now dnsmasq

sudo tee /etc/dnsmasq.d/dns-lab.conf > /dev/null <<'EOF'
port=53
interface=enp65s0f0np0
listen-address=10.10.1.2
bind-interfaces
no-resolv
no-hosts
address=/lab.test/192.0.2.10
EOF

sudo dnsmasq --test
sudo systemctl restart dnsmasq
sudo systemctl status dnsmasq --no-pager

FOR NFTABLES METRICS 
sudo nft add table inet dns_filter
sudo nft add chain inet dns_filter input { type filter hook input priority 0 \; }
sudo nft add rule inet dns_filter input udp dport 53 meter flood_limit { ip saddr limit rate over 1000/second burst 100 packets } drop
